# Deployment Guide

This guide covers deploying your blockchain VM to various network environments, from local testing to production mainnet deployment.

## Deployment Overview

### Deployment Types

| Type | Purpose | Validators | Network | Persistence |

| Mainnet | Production | 20+ | Public | Required |
| Sovereign L1 | Independent | Custom | Private | Required |

## Local Deployment

### Quick Start

```bash
# Build your VM
go build -o myvm ./cmd/myvm

# Create local network
luxcli network create local \
  --num-nodes 5 \
  --vm-binary ./myvm \
  --vm-genesis genesis.json

# Deploy blockchain
luxcli blockchain deploy \
  --network local \
  --vm myvm \
  --name "MyChain"

# Get RPC endpoint
luxcli blockchain describe MyChain --network local
```

### Mainnet Deployment Process

#### Step 1: Validator Setup

```bash
#!/bin/bash

# Install node software
wget https://github.com/luxfi/node/releases/latest/luxd
chmod +x luxd
sudo mv luxd /usr/local/bin/

# Create data directory
sudo mkdir -p /var/lib/luxd
sudo chown $USER:$USER /var/lib/luxd

# Generate node certificate
luxd --generate-staking-cert \
  --staking-tls-cert-file=/var/lib/luxd/staking/cert.pem \
  --staking-tls-key-file=/var/lib/luxd/staking/key.pem

# Create systemd service
sudo tee /etc/systemd/system/luxd.service > /dev/null <<EOF
[Unit]
Description=Lux Node
After=network.target

[Service]
Type=simple
User=$USER
ExecStart=/usr/local/bin/luxd \
  --network-id=1 \
  --data-dir=/var/lib/luxd \
  --api-admin-enabled=false \
  --api-ipcs-enabled=false \
  --api-keystore-enabled=false \
  --http-host=0.0.0.0 \
  --http-port=9650 \
  --staking-port=9651 \
  --public-ip=$(curl -s ifconfig.me) \
  --snow-mixed-query-num-push-non-vdr=10 \
  --log-level=info

Restart=on-failure
RestartSec=10

[Install]
WantedBy=multi-user.target
EOF

# Start service
sudo systemctl daemon-reload
sudo systemctl enable luxd
sudo systemctl start luxd
```

#### Step 2: Deploy Subnet

```go
// Deploy subnet to mainnet
func deployToMainnet(config *MainnetConfig) error {
    // Connect to mainnet
    client, err := NewClient(config.Endpoint)
    if err != nil {
        return err
    }

    // Create subnet
    subnetTx := &CreateSubnetTx{
        Owner: config.Owner,
        Threshold: config.Threshold,
    }

    subnetID, err := client.CreateSubnet(subnetTx)
    if err != nil {
        return fmt.Errorf("failed to create subnet: %w", err)
    }

    log.Printf("Subnet created: %s", subnetID)

    // Add validators
    for _, validator := range config.Validators {
        addValidatorTx := &AddSubnetValidatorTx{
            SubnetID: subnetID,
            NodeID:   validator.NodeID,
            Weight:   validator.Weight,
            StartTime: time.Now(),
            EndTime:   time.Now().Add(365 * 24 * time.Hour),
        }

        if err := client.AddSubnetValidator(addValidatorTx); err != nil {
            return fmt.Errorf("failed to add validator %s: %w",
                validator.NodeID, err)
        }

        log.Printf("Added validator: %s", validator.NodeID)
    }

    // Deploy blockchain
    createChainTx := &CreateChainTx{
        SubnetID:    subnetID,
        ChainName:   config.ChainName,
        VMID:        config.VMID,
        FxIDs:       config.FxIDs,
        GenesisData: config.Genesis,
    }

    chainID, err := client.CreateBlockchain(createChainTx)
    if err != nil {
        return fmt.Errorf("failed to create blockchain: %w", err)
    }

    log.Printf("Blockchain deployed: %s", chainID)

    // Wait for validators to sync
    if err := waitForValidatorSync(client, chainID); err != nil {
        return err
    }

    log.Printf("Mainnet deployment complete!")
    log.Printf("Chain ID: %s", chainID)
    log.Printf("RPC Endpoint: %s/ext/bc/%s/rpc", config.Endpoint, chainID)

    return nil
}
```

### Security Hardening

```bash
#!/bin/bash

# Firewall configuration
sudo ufw default deny incoming
sudo ufw default allow outgoing

sudo ufw allow 9650/tcp  # HTTP API

enabled = true

logpath = /var/log/luxd/luxd.log

sudo systemctl restart fail2ban

# SSL/TLS setup with Let's Encrypt
sudo apt-get install certbot nginx
sudo certbot certonly --nginx -d api.mychain.com

# Nginx reverse proxy
sudo tee /etc/nginx/sites-available/luxd > /dev/null <<EOF
server {
    listen 443 ssl http2;
    server_name api.mychain.com;

    ssl_certificate /etc/letsencrypt/live/api.mychain.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/api.mychain.com/privkey.pem;

        proxy_pass http://localhost:9650;

        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto \$scheme;
    }
}
EOF

sudo nginx -s reload
```

### Deploy Sovereign L1

```go
func deploySovereignL1(config *SovereignConfig) error {
    // Initialize genesis
    genesis := &Genesis{
        NetworkID:     config.NetworkID,
        Validators:    config.ValidatorSet,
        InitialSupply: config.InitialSupply,
        StartTime:     time.Now(),
    }

    // Create bootstrap nodes
    bootstrapNodes := make([]*Node, len(config.BootstrapNodes))
    for i, endpoint := range config.BootstrapNodes {
        node, err := createBootstrapNode(endpoint, genesis)
        if err != nil {
            return err
        }
        bootstrapNodes[i] = node
    }

    // Deploy validator nodes
    for _, validator := range config.ValidatorSet {
        if err := deployValidatorNode(validator, genesis, bootstrapNodes); err != nil {
            return fmt.Errorf("failed to deploy validator %s: %w",
                validator.NodeID, err)
        }
    }

    // Deploy API nodes
    apiEndpoints := deployAPINodes(config, bootstrapNodes)

    log.Printf("Sovereign L1 deployed successfully!")
    log.Printf("Network ID: %d", config.NetworkID)
    log.Printf("Validators: %d", len(config.ValidatorSet))
    log.Printf("API Endpoints: %v", apiEndpoints)

    return nil
}
```

### Health Monitoring Script

```go
// Health monitoring service
type HealthMonitor struct {
    nodes    []*Node
    interval time.Duration
    alerter  Alerter
}

func (hm *HealthMonitor) Start() {
    ticker := time.NewTicker(hm.interval)
    defer ticker.Stop()

    for range ticker.C {
        for _, node := range hm.nodes {
            if err := hm.checkNode(node); err != nil {
                hm.alerter.Alert(AlertCritical,
                    fmt.Sprintf("Node %s unhealthy: %v", node.ID, err))
            }
        }
    }
}

func (hm *HealthMonitor) checkNode(node *Node) error {
    // Check if node is reachable
    health, err := node.HealthCheck()
    if err != nil {
        return err
    }

    // Check sync status
    if !health.IsSynced {
        return fmt.Errorf("node not synced")
    }

    // Check peer count
    if health.PeerCount < 3 {
        return fmt.Errorf("low peer count: %d", health.PeerCount)
    }

    // Check disk space
    if health.DiskUsage > 90 {
        return fmt.Errorf("high disk usage: %d%%", health.DiskUsage)
    }

    return nil
}
```

## Backup and Recovery

### Backup Strategy

```bash
#!/bin/bash

# Backup script
BACKUP_DIR="/backup/luxd"
DATA_DIR="/var/lib/luxd"
DATE=$(date +%Y%m%d_%H%M%S)

# Stop node
sudo systemctl stop luxd

# Create backup
tar -czf "$BACKUP_DIR/luxd_backup_$DATE.tar.gz" \
  "$DATA_DIR/db" \
  "$DATA_DIR/staking" \
  "$DATA_DIR/configs"

# Start node
sudo systemctl start luxd

# Upload to S3
aws s3 cp "$BACKUP_DIR/luxd_backup_$DATE.tar.gz" \
  s3://mybackups/luxd/

# Clean old backups (keep last 30 days)
find "$BACKUP_DIR" -name "*.tar.gz" -mtime +30 -delete
```

### Disaster Recovery

```go
// Disaster recovery procedure
func recoverNode(backupPath string) error {
    // Stop node if running
    if err := stopNode(); err != nil {
        return err
    }

    // Restore from backup
    if err := restoreBackup(backupPath); err != nil {
        return err
    }

    // Verify database integrity
    if err := verifyDatabase(); err != nil {
        return err
    }

    // Resync with network
    if err := startNode(ResyncMode); err != nil {
        return err
    }

    // Wait for sync
    if err := waitForSync(); err != nil {
        return err
    }

    log.Println("Node recovered successfully")
    return nil
}
```

## Upgrade Procedures

### Rolling Upgrade

```bash
#!/bin/bash

# Rolling upgrade script
NODES=("node1.example.com" "node2.example.com" "node3.example.com")
NEW_VERSION="v1.2.0"

for NODE in "${NODES[@]}"; do
    echo "Upgrading $NODE to $NEW_VERSION"

    # Drain node connections
    ssh $NODE "luxcli node drain"

    # Wait for drain
    sleep 30

    # Stop node
    ssh $NODE "sudo systemctl stop luxd"

    # Backup current version
    ssh $NODE "cp /usr/local/bin/luxd /usr/local/bin/luxd.bak"

    # Download new version
    ssh $NODE "wget https://github.com/luxfi/node/releases/download/$NEW_VERSION/luxd"
    ssh $NODE "chmod +x luxd && sudo mv luxd /usr/local/bin/"

    # Start node
    ssh $NODE "sudo systemctl start luxd"

    # Wait for sync
    echo "Waiting for $NODE to sync..."
    sleep 120

    # Verify health
    if ! ssh $NODE "luxcli node health"; then
        echo "ERROR: $NODE unhealthy after upgrade"
        exit 1
    fi

    echo "$NODE upgraded successfully"
done

echo "Rolling upgrade complete!"
```

## Troubleshooting

### Common Issues

**Issue**: Node not syncing
```bash
# Check peer connections
curl -X POST --data '{
    "jsonrpc":"2.0",
    "id":1,
    "method":"info.peers"
}' -H 'content-type:application/json;' localhost:9650/ext/info

# Check bootstrap status
curl -X POST --data '{
    "jsonrpc":"2.0",
    "id":1,
    "method":"info.isBootstrapped",
    "params": {"chain":"X"}
}' -H 'content-type:application/json;' localhost:9650/ext/info
```

**Issue**: High memory usage
```bash
# Check memory stats
cat /proc/$(pgrep luxd)/status | grep VmRSS

# Restart with memory limit
systemctl edit luxd
# Add: MemoryMax=8G
systemctl restart luxd
```

**Issue**: Database corruption
```bash
# Stop node
systemctl stop luxd

# Backup corrupted database
mv /var/lib/luxd/db /var/lib/luxd/db.corrupted

# Resync from network
systemctl start luxd
```
